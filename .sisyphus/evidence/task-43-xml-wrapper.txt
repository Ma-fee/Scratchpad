XML Schema:
-----------
The XML wrapper generates standardized XML format:

<?xml version="1.0" encoding="UTF-8"?>
<resource type="text|image|binary|directory">
  <uri>scratchpad://...</uri>
  <session_id>...</session_id>
  <path>...</path>
  <metadata>
    <size>...</size>
    <mimetype>...</mimetype>
    <modified>...</modified>
    <created>...</created>
    <content_type>...</content_type>
    <extension>...</extension>
    <is_directory>false</is_directory>
  </metadata>
  <content encoding="base64|plain">
    <is_binary>false</is_binary>
    <size_bytes>...</size_bytes>
    <data>...</data>
  </content>
  <success>true</success>
  <error>...</error>  <!-- if error -->
  <subscription_info>
    <is_subscribed>false</is_subscribed>
    ...
  </subscription_info>
  <capabilities>
    <capability>read</capability>
    <capability>read_chunk</capability>
    ...
  </capabilities>
</resource>
